eight};
  }

  function roundRect(x,y,w,h,r){
    ctx.beginPath();
    if(ctx.roundRect) ctx.roundRect(x,y,w,h,r);
    else { ctx.rect(x,y,w,h); }
  }

  function worldView(P,rect){
    const ca=Math.cos(state.viewAz),sa=Math.sin(state.viewAz);
    const ce=Math.cos(state.viewEl),se=Math.sin(state.viewEl);
    const shift=state.task==='headPose'?[P[0],P[1],P[2]+3.0]:[P[0],P[1],P[2]+3.25];
    const x=ca*shift[0]-sa*shift[2];
    const z=sa*shift[0]+ca*shift[2];
    const y=ce*shift[1]-se*z;
    return [rect.x+rect.w*.5+x*state.viewZoom,rect.y+rect.h*.56-y*state.viewZoom];
  }

  function line3(A,B,rect,color,width=1,dash=[]){
    const a=worldView(A,rect), b=worldView(B,rect);
    ctx.save();
    ctx.strokeStyle=color; ctx.lineWidth=width; ctx.setLineDash(dash);
    ctx.beginPath(); ctx.moveTo(a[0],a[1]); ctx.lineTo(b[0],b[1]); ctx.stroke();
    ctx.restore();
  }

  function drawArrow3(A,B,rect,color,width=2){
    line3(A,B,rect,color,width);
    const a=worldView(A,rect),b=worldView(B,rect);
    const dx=b[0]-a[0],dy=b[1]-a[1],m=Math.hypot(dx,dy)||1;
    const ux=dx/m,uy=dy/m;
    ctx.save();ctx.fillStyle=color;ctx.beginPath();
    ctx.moveTo(b[0],b[1]);
    ctx.lineTo(b[0]-ux*11-uy*5,b[1]-uy*11+ux*5);
    ctx.lineTo(b[0]-ux*11+uy*5,b[1]-uy*11-ux*5);
    ctx.closePath();ctx.fill();ctx.restore();
  }

  function drawAxes(rect){
    const O=[0,0,0];
    drawArrow3(O,[1.25,0,0],rect,css('--red'),3);
    drawArrow3(O,[0,1.25,0],rect,css('--green'),3);
    drawArrow3(O,[0,0,1.25],rect,css('--blue'),3);
    [['X',[1.36,0,0],css('--red')],['Y',[0,1.36,0],css('--green')],['Z',[0,0,1.36],css('--blue')]].forEach(([label,P,c])=>{
      const q=worldView(P,rect);ctx.save();ctx.fillStyle=c;ctx.font='800 12px system-ui';ctx.fillText(label,q[0],q[1]);ctx.restore();
    });
  }

  function drawCamera(rig,cam,rect,color,dashed,label,alpha=1){
    const p=cameraPose(rig,cam);
    const C=p.slice(0,3);
    const corners=[[-.68,-.49,1.18],[.68,-.49,1.18],[.68,.49,1.18],[-.68,.49,1.18]].map(q=>localToWorld(q,p));
    ctx.save();ctx.globalAlpha=alpha;ctx.restore();
    corners.forEach(q=>line3(C,q,rect,color,2,dashed?[7,5]:[]));
    for(let i=0;i<4;i++) line3(corners[i],corners[(i+1)%4],rect,color,2,dashed?[7,5]:[]);
    const q=worldView(C,rect);
    ctx.save();ctx.globalAlpha=alpha;ctx.fillStyle=color;ctx.beginPath();ctx.arc(q[0],q[1],6,0,Math.PI*2);ctx.fill();ctx.font='800 11px system-ui';ctx.fillText(label,q[0]+8,q[1]-7);ctx.restore();
  }

  function drawTaskShape(rect,estimatedPose=null,estimated=false){
    const t=task();
    const drawPoint=P=>estimatedPose?localToWorld(P,estimatedPose):P;
    const edgeColor=estimated?css('--green'):css('--border');

    if(state.task==='eyeBoard'){
      const corners=[[-2.55,-1.62,0],[2.55,-1.62,0],[2.55,1.62,0],[-2.55,1.62,0]].map(drawPoint);
      const poly=corners.map(P=>worldView(P,rect));
      ctx.save();ctx.fillStyle=estimated?'rgba(5,150,105,.06)':'rgba(37,99,235,.055)';ctx.strokeStyle=edgeColor;ctx.lineWidth=estimated?2:1.5;ctx.setLineDash(estimated?[7,5]:[]);ctx.beginPath();ctx.moveTo(...poly[0]);poly.slice(1).forEach(q=>ctx.lineTo(...q));ctx.closePath();ctx.fill();ctx.stroke();ctx.restore();
      [-.81,0,.81].forEach(y=>line3(drawPoint([-2.55,y,0]),drawPoint([2.55,y,0]),rect,edgeColor,.8,estimated?[5,5]:[]));
      [-1.27,0,1.27].forEach(x=>line3(drawPoint([x,-1.62,0]),drawPoint([x,1.62,0]),rect,edgeColor,.8,estimated?[5,5]:[]));
    }

    t.edges.forEach(([a,b])=>line3(drawPoint(t.points[a]),drawPoint(t.points[b]),rect,edgeColor,estimated?1.5:1.2,estimated?[6,5]:[]));
  }

  function drawTargetAndGaze(rect,estimate){
    if(state.stage!==9 || state.task!=='eyeBoard' || !task().target) return;
    const target=task().target;
    const q=worldView(target,rect);
    ctx.save();ctx.fillStyle=css('--yellow');ctx.strokeStyle=css('--panel');ctx.lineWidth=2;ctx.beginPath();
    for(let k=0;k<10;k++){
      const a=-Math.PI/2+k*Math.PI/5;
      const r=k%2===0?9:4;
      const x=q[0]+Math.cos(a)*r,y=q[1]+Math.sin(a)*r;
      if(k===0)ctx.moveTo(x,y);else ctx.lineTo(x,y);
    }
    ctx.closePath();ctx.fill();ctx.stroke();ctx.font='800 11px system-ui';ctx.fillStyle=css('--yellow');ctx.fillText('fixation target',q[0]+12,q[1]-8);ctx.restore();

    const rig=estimate||trueRigPose();
    const eye=localToWorld([0,.18,-.38],rig);
    drawArrow3(eye,target,rect,css('--yellow'),3);
    const e=worldView(eye,rect);
    ctx.save();ctx.fillStyle=css('--yellow');ctx.beginPath();ctx.ellipse(e[0],e[1],9,6,0,0,Math.PI*2);ctx.fill();ctx.fillStyle=css('--panel');ctx.beginPath();ctx.arc(e[0]+2,e[1],2.5,0,Math.PI*2);ctx.fill();ctx.restore();
  }

  function drawHeadUse(rect){
    if(state.stage!==9 || state.task!=='headPose') return;
    const eyeCenter=[0,.52,.05];
    const headForward=[.15,.32,-2.0];
    drawArrow3(eyeCenter,headForward,rect,css('--yellow'),3);
    const q=worldView(headForward,rect);
    ctx.save();ctx.fillStyle=css('--yellow');ctx.font='800 11px system-ui';ctx.fillText('separate gaze ray',q[0]+7,q[1]-5);ctx.restore();
  }

  function drawWorldPanel(rect,now,estimate){
    ctx.save();ctx.fillStyle=css('--panel');ctx.strokeStyle=css('--border');roundRect(rect.x,rect.y,rect.w,rect.h,16);ctx.fill();ctx.stroke();
    ctx.fillStyle=css('--text');ctx.font='800 14px system-ui';ctx.fillText('3D model/world view',rect.x+14,rect.y+24);
    ctx.fillStyle=css('--muted');ctx.font='12px system-ui';ctx.fillText('Drag to orbit · wheel to zoom',rect.x+14,rect.y+42);

    if(state.stage>=1) drawAxes(rect);
    drawTaskShape(rect);

    const trueRig=trueRigPose();
    const showTrue=state.stage!==6 || Math.floor(now/450)%2===0;
    if(showTrue){
      drawCamera(trueRig,0,rect,css('--blue'),false,state.cameraCount===2?'C1 true':'true camera',state.stage===6?.55:1);
      if(state.cameraCount===2){
        drawCamera(trueRig,1,rect,css('--purple'),false,'C2 true',state.stage===6?.55:1);
        line3(cameraPose(trueRig,0).slice(0,3),cameraPose(trueRig,1).slice(0,3),rect,css('--purple'),4);
      }
    }

    if(state.stage>=4){
      activeCameras().forEach(cam=>{
        const C=cameraPose(trueRig,cam).slice(0,3);
        points().forEach((P,i)=>{
          if(!observation(i,cam)) return;
          const rayColor=cam===0?css('--blue'):css('--purple');
          line3(C,P,rect,rayColor,1,[4,6]);
          if(state.rays){
            const phase=(now*.00034*state.speed+i/points().length+cam*.42)%1;
            const Q=add(C,scale(sub(P,C),phase));
            const s=worldView(Q,rect);
            ctx.save();ctx.fillStyle=css(COLOR_VARS[i%COLOR_VARS.length]);ctx.beginPath();ctx.arc(s[0],s[1],3.7,0,Math.PI*2);ctx.fill();ctx.restore();
          }
        });
      });
    }

    const highlightIndex=Math.floor(now/900)%points().length;
    points().forEach((P,i)=>{
      const q=worldView(P,rect);
      const pulse=state.stage===2 && i===highlightIndex ? 1.45 : 1+.08*Math.sin(now*.005+i);
      ctx.save();ctx.fillStyle=css(COLOR_VARS[i%COLOR_VARS.length]);ctx.strokeStyle=css('--panel');ctx.lineWidth=2;ctx.beginPath();ctx.arc(q[0],q[1],5.8*pulse,0,Math.PI*2);ctx.fill();ctx.stroke();ctx.fillStyle=css('--text');ctx.font='800 10px system-ui';ctx.fillText(labels()[i],q[0]+8,q[1]-7);ctx.restore();
      if(state.stage===2 && i===highlightIndex){
        const text=`${labels()[i]} = (${P.map(v=>v.toFixed(2)).join(', ')})`;
        ctx.save();ctx.font='800 11px system-ui';const tw=ctx.measureText(text).width;ctx.fillStyle=css('--blue-soft');roundRect(q[0]+10,q[1]+7,tw+16,25,8);ctx.fill();ctx.fillStyle=css('--text');ctx.fillText(text,q[0]+18,q[1]+24);ctx.restore();
      }
    });

    if(state.stage===6){
      const C=worldView(trueRig.slice(0,3),rect);
      const r=28+6*Math.sin(now*.004);
      ctx.save();ctx.fillStyle=css('--orange');ctx.font='900 28px system-ui';ctx.fillText('?',C[0]+Math.cos(now*.002)*r,C[1]+Math.sin(now*.002)*r);ctx.fillText('R,t ?',C[0]-24,C[1]-34);ctx.restore();
    }

    if(state.stage>=7 && estimate){
      drawCamera(estimate,0,rect,css('--green'),true,state.stage===7?'initial guess':'estimated C1');
      if(state.cameraCount===2){
        drawCamera(estimate,1,rect,css('--green'),true,state.stage===7?'initial C2':'estimated C2');
        line3(cameraPose(estimate,0).slice(0,3),cameraPose(estimate,1).slice(0,3),rect,css('--green'),3,[7,5]);
      }
    }

    drawTargetAndGaze(rect,estimate);
    drawHeadUse(rect);
    ctx.restore();
  }

  function imageMap(q,rect){
    return [rect.x+q[0]/IMG_W*rect.w,rect.y+40+q[1]/IMG_H*(rect.h-48)];
  }

  function drawImageEdges(rect,cam,predictedRig=null,color=null,dash=[]){
    const coords=points().map((P,i)=>{
      const q=predictedRig?project(P,predictedRig,cam):observation(i,cam);
      return q?imageMap(q,rect):null;
    });
    ctx.save();ctx.strokeStyle=color||css('--border');ctx.lineWidth=1.1;ctx.setLineDash(dash);
    task().edges.forEach(([a,b])=>{
      if(!coords[a]||!coords[b])return;
      ctx.beginPath();ctx.moveTo(...coords[a]);ctx.lineTo(...coords[b]);ctx.stroke();
    });
    ctx.restore();
  }
