  function rotation(p){
    const yaw=p[3], pitch=p[4], roll=p[5];
    const cy=Math.cos(yaw), sy=Math.sin(yaw);
    const cp=Math.cos(pitch), sp=Math.sin(pitch);
    const cr=Math.cos(roll), sr=Math.sin(roll);
    const Ry=[[cy,0,sy],[0,1,0],[-sy,0,cy]];
    const Rx=[[1,0,0],[0,cp,-sp],[0,sp,cp]];
    const Rz=[[cr,-sr,0],[sr,cr,0],[0,0,1]];
    return matMul(Rz,matMul(Rx,Ry));
  }

  function observation(i,cam){

  }

  function activeCameras(){ return state.cameraCount===2 ? [0,1] : [0]; }

  function residualPairs(p){
    const out=[];
    for(const cam of activeCameras()){
      points().forEach((P,i)=>{
        const o=observation(i,cam);
        const q=project(P,p,cam);
        if(o && q) out.push({cam,i,dx:q[0]-o[0],dy:q[1]-o[1]});
      });
    }
    return out;
  }

  function residualVector(p){
    return residualPairs(p).flatMap(r=>[r.dx,r.dy]);
  }

  function huberWeight(error,delta=13){
    if(!state.robust || error<=delta) return 1;
    return delta/Math.max(delta,error);
  }

  function weightedCost(p){
    return residualPairs(p).reduce((sum,r)=>{
      const e=Math.hypot(r.dx,r.dy);
      return sum+huberWeight(e)*(r.dx*r.dx+r.dy*r.dy);
    },0);
  }
