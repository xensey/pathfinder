near6(A,b){
    const n=6;
    const M=A.map((r,i)=>r.concat(b[i]));
    for(let c=0;c<n;c++){
      let pivot=c;
      for(let r=c+1;r<n;r++) if(Math.abs(M[r][c])>Math.abs(M[pivot][c])) pivot=r;
      if(Math.abs(M[pivot][c])<1e-11) return null;
      [M[c],M[pivot]]=[M[pivot],M[c]];
      const d=M[c][c];
      for(let j=c;j<=n;j++) M[c][j]/=d;
      for(let r=0;r<n;r++) if(r!==c){
        const f=M[r][c];
        for(let j=c;j<=n;j++) M[r][j]-=f*M[c][j];
      }
    }
    return M.map(r=>r[n]);
  }

  function initialPose(){
    return add(trueRigPose(),[1.15,-.78,1.15,13*DEG,-10*DEG,8*DEG]);
  }

  function prepareSolver(){
    let p=initialPose();
    const history=[{pose:p.slice(),rmse:reprojectionRmse(p),cost:weightedCost(p)}];
    const eps=[1e-3,1e-3,1e-3,1e-4,1e-4,1e-4];
    let lambda=2e-2;

    for(let iter=0;iter<38;iter++){
      const pairs=residualPairs(p);
      const r=pairs.flatMap(x=>[x.dx,x.dy]);
      const J=Array.from({length:r.length},()=>Array(6).fill(0));

      for(let j=0;j<6;j++){
        const q=p.slice(); q[j]+=eps[j];
        const rr=residualVector(q);
        if(rr.length!==r.length) continue;
        for(let row=0;row<r.length;row++) J[row][j]=(rr[row]-r[row])/eps[j];
      }

      const A=Array.from({length:6},()=>Array(6).fill(0));
      const b=Array(6).fill(0);
      for(let pairIndex=0;pairIndex<pairs.length;pairIndex++){
        const pair=pairs[pairIndex];
        const w=huberWeight(Math.hypot(pair.dx,pair.dy));
        for(let axis=0;axis<2;axis++){
          const row=pairIndex*2+axis;
          for(let a=0;a<6;a++){
            b[a]+=w*J[row][a]*r[row];
            for(let c=0;c<6;c++) A[a][c]+=w*J[row][a]*J[row][c];
          }
        }
      }
      for(let j=0;j<6;j++) A[j][j]+=lambda;
      const d=solveLinear6(A,b);
      if(!d) break;
      const candidate=p.map((v,j)=>v-d[j]);
      const currentCost=weightedCost(p);
      const candidateCost=weightedCost(candidate);
      if(candidateCost < currentCost){
        p=candidate;
        lambda=Math.max(1e-7,lambda*.45);
        history.push({pose:p.slice(),rmse:reprojectionRmse(p),cost:candidateCost});
        if(mag(d)<1e-7) break;
      }else{
        lambda=Math.min(1e7,lambda*5.5);
      }
    }
    state.history=history;
  }

  function rotationError(a,b){
    const Q=matMul(rotation(a),transpose(rotation(b)));
    const v=(Q[0][0]+Q[1][1]+Q[2][2]-1)/2;
    return Math.acos(Math.max(-1,Math.min(1,v)))/DEG;
  }

  function currentEstimate(now){
    if(state.stage<7) return null;
    if(!state.history.length) prepareSolver();
    if(state.stage===7) return state.history[0].pose;
    if(state.stage===9 && !state.solving) return state.history[state.history.length-1].pose;
    if(!state.solving) return state.history[state.history.length-1].pose;

    const stepMs=760/Math.max(.1,state.speed);
    const rawF=(now-state.solveStart)/stepMs;
    const f=Number.isFinite(rawF)?Math.max(0,rawF):0;
    const last=state.history.length-1;
    const i=Math.max(0,Math.min(last,Math.floor(f)));
    if(i>=last || !state.history[i] || !state.history[i+1]){
      state.solving=false;
      if(state.stage===8) setTimeout(()=>setStage(9,false),280);
      return state.history[last].pose;
    }
    const t=Math.max(0,Math.min(1,f-i));
    const smooth=t*t*(3-2*t);
    return mix(state.history[i].pose,state.history[i+1].pose,smooth);
  }

  function currentIteration(now){
    if(!state.history.length) return 0;
    if(!state.solving) return state.stage===7?0:state.history.length-1;
    return Math.min(state.history.length-1,Math.floor((now-state.solveStart)/(760/state.speed)));
  }
