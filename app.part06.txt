  function computeDomains() {
    const byId=Object.fromEntries(state.results.map(r=>[r.id,r]));
    const pct=(ids)=>{
      const rs=ids.map(id=>byId[id]).filter(Boolean); const s=rs.reduce((a,r)=>a+r.score,0), m=rs.reduce((a,r)=>a+r.max,0); return m?Math.round(s/m*100):0;
    };
    const calibrationEntries=state.results.filter(r=>r.details&&typeof r.details.confidence==='number');
    const cal = calibrationEntries.length ? Math.round(100-calibrationEntries.reduce((a,r)=>a+Math.abs(r.details.confidence-(r.score/r.max*100)),0)/calibrationEntries.length) : 0;
    return [
      {name:'Рабочая и отсроченная память',value:pct(['reverse','delayed','prospective'])},
      {name:'Связывание и отношения',value:pct(['relations','matrix','logic'])},
      {name:'Индукция и гипотезы',value:pct(['number','hypothesis'])},
      {name:'Переключение правила',value:pct(['switch'])},
      {name:'Поиск информации и планирование',value:pct(['info','planning'])},
      {name:'Контроль ошибок',value:pct(['monitoring'])},
      {name:'Калибровка уверенности',value:Math.max(0,Math.min(100,cal))}
    ];
  }

  function interpretation(domains) {
    const texts=[];
    const low=domains.filter(d=>d.value<50 && d.name!=='Калибровка уверенности');
    const strong=domains.filter(d=>d.value>=80 && d.name!=='Калибровка уверенности');
    if(strong.length) texts.push(`Наиболее устойчивые результаты в этой сессии: ${strong.map(x=>x.name.toLowerCase()).join(', ')}.`);
    if(low.length) texts.push(`Зоны с наибольшим числом ошибок в этой сессии: ${low.map(x=>x.name.toLowerCase()).join(', ')}.`);
    const cal=domains.find(d=>d.name==='Калибровка уверенности')?.value ?? 0;
    if(cal<55) texts.push('Уверенность заметно расходилась с фактической точностью; полезно отдельно смотреть случаи высокой уверенности при ошибочном ответе.');
    const info=state.results.find(r=>r.id==='info');
    if(info && info.details.essentialN<3) texts.push('В задаче с ограниченным числом запросов были выбраны не все критически необходимые параметры.');
    const hyp=state.results.find(r=>r.id==='hypothesis');
    if(hyp && !hyp.details.hasNegativeProbe) texts.push('При проверке скрытого правила почти не использовались опровергающие примеры — возможна склонность подтверждать первую гипотезу.');
    const sw=state.results.find(r=>r.id==='switch');
    if(sw && sw.details.postSwitchErrors>=2) texts.push('После смены правила возникли повторные ошибки в первых пробах; это может отражать инерцию предыдущей установки в рамках данного задания.');
    if(!texts.length) texts.push('Профиль без выраженного локального провала; для интерпретации важны повторная параллельная версия и сравнение с контрольной группой в одинаковых условиях.');
    return texts;
  }
