cingMode: 'user', width: { ideal: 640 }, height: { ideal: 480 }, frameRate: { ideal: 15, max: 20 } },
        audio: false
      });
      state.camera.stream = stream;
      state.camera.status = 'ready';
      const preview = document.getElementById('cameraPreview');
      if (preview) {
        preview.srcObject = stream;
        await preview.play().catch(()=>{});
      }
    } catch (err) {
      state.camera.status = err?.name === 'NotAllowedError' ? 'denied' : 'error';
      state.camera.error = `${err?.name || 'Error'}: ${err?.message || String(err)}`;
      if (errorBox) { errorBox.textContent = `Камера недоступна: ${err?.message || err}`; errorBox.classList.remove('hidden'); }
    }
    updateCameraUI();
  }

  function chooseVideoMimeType() {
    const candidates = ['video/webm;codecs=vp9','video/webm;codecs=vp8','video/webm','video/mp4'];
    return candidates.find(t => typeof MediaRecorder !== 'undefined' && MediaRecorder.isTypeSupported?.(t)) || '';
  }

  async function startCameraRecording() {
    if (!state.camera.stream || state.camera.status !== 'ready') return false;
    try {
      state.camera.chunks = [];
      state.camera.mimeType = chooseVideoMimeType();
      const options = { videoBitsPerSecond: 350000 };
      if (state.camera.mimeType) options.mimeType = state.camera.mimeType;
      const recorder = new MediaRecorder(state.camera.stream, options);
      state.camera.recorder = recorder;
      recorder.addEventListener('dataavailable', event => { if (event.data?.size) state.camera.chunks.push(event.data); });
      recorder.addEventListener('error', event => {
        state.camera.error = event?.error?.message || 'MediaRecorder error';
        state.camera.status = 'error';
      });
      recorder.start(1000);
      state.camera.startedAt = new Date().toISOString();
      state.camera.status = 'recording';
      return true;
    } catch (err) {
      state.camera.error = `${err?.name || 'Error'}: ${err?.message || String(err)}`;
      state.camera.status = 'error';
      return false;
    }
  }

  async function stopCameraRecording() {
    const cam = state.camera;
    if (cam.recorder && cam.recorder.state !== 'inactive') {
      await new Promise(resolve => {
        const finish = () => resolve();
        cam.recorder.addEventListener('stop', finish, { once: true });
        try { cam.recorder.stop(); } catch (_) { resolve(); }
        setTimeout(resolve, 2500);
      });
    }
    if (!cam.blob && cam.chunks.length) {
      cam.blob = new Blob(cam.chunks, { type: cam.mimeType || cam.chunks[0]?.type || 'video/webm' });
    }
    cam.stream?.getTracks?.().forEach(track => track.stop());
    cam.stoppedAt = new Date().toISOString();
    if (cam.status === 'recording') cam.status = cam.blob ? 'recorded' : 'stopped';
  }

  function stopCameraPreview() {
    state.camera.stream?.getTracks?.().forEach(track => track.stop());
    state.camera.stream = null;
  }

    state.prospective.instructionSeen = true;
    if (useCamera) await startCameraRecording();
    else { stopCameraPreview(); state.camera.status = 'skipped'; }
    task1Encoding();
  }
