  function buildStageTabs(){
    STAGES.forEach((s,i)=>{
      const b=document.createElement('button');
      b.type='button';
      b.className='stage-btn';
      b.innerHTML=`<span class="stage-num">${i+1}</span><span class="stage-label">${s.short}</span>`;
      b.addEventListener('click',()=>setStage(i,true));
      $('stageTabs').appendChild(b);
    });
  }

  function buildSliders(){
    sliderDefs.forEach(([key,label,min,max,step,unit])=>{
      const row=document.createElement('label');
      row.className='slider-row';
      row.id=`${key}Row`;
      row.innerHTML=`<span class="slider-top"><span>${label}</span><output id="${key}Out"></output></span><input id="${key}Slider" type="range" min="${min}" max="${max}" step="${step}" />`;
      $('sliderList').appendChild(row);
      $(`${key}Slider`).addEventListener('input',e=>{
        state[key]=+e.target.value;
        invalidateSolution();
        syncUI();
      });
    });
  }

  function invalidateSolution(){
    state.solving=false;
    state.history=[];
  }

  function setStage(index,autoSolve=true){
    state.stage=Math.max(0,Math.min(STAGES.length-1,index));
    if(state.stage>=7 && !state.history.length) prepareSolver();
    if(state.stage===8 && autoSolve){
      state.solving=true;
      state.solveStart=performance.now();
    }else if(state.stage!==8){
      state.solving=false;
    }
    syncUI();
  }

  function visibleMatchCount(){
    let n=0;
    activeCameras().forEach(cam=>points().forEach((_,i)=>{ if(observation(i,cam)) n++; }));
    return n;
  }

  function updatePointTable(){
    $('pointTableBody').innerHTML=points().map((p,i)=>`<tr><td><strong>${labels()[i]}</strong></td><td>${p[0].toFixed(2)}</td><td>${p[1].toFixed(2)}</td><td>${p[2].toFixed(2)}</td></tr>`).join('');
  }

  function updatePipeline(){
    const items=[
      'Load known 3D model coordinates',
      'Calibrate camera intrinsics'+(state.cameraCount===2?' and rig extrinsics':''),
      'Detect 2D landmarks in the current image'+(state.cameraCount===2?'s':''),
      'Match every 2D observation to its 3D identity',
      'Solve and refine one 6-DoF pose',
      state.task==='eyeBoard'?'Use the recovered 3D target for gaze calibration':state.task==='headPose'?'Use head pose to normalize/transform gaze':'Use pose in the downstream CV system'
    ];
    const stageToPipe=[0,0,0,1,2,3,4,4,4,5][state.stage];
    $('pipeline').innerHTML=items.map((x,i)=>`<div class="pipe-step ${i===stageToPipe?'active':''}" data-step="${i+1}">${x}</div>`).join('');
  }

  function syncUI(){
    const t=task();
    const copy=stageCopy();
    [...$('stageTabs').children].forEach((b,i)=>b.classList.toggle('active',i===state.stage));
    $('stageBadge').textContent=`Stage ${state.stage+1} of ${STAGES.length}`;
    $('stageTitle').textContent=copy.title;
    $('stageText').textContent=copy.text;
    $('knownStage').textContent=copy.known;
    $('unknownStage').textContent=copy.unknown;
    $('equation').textContent=copy.equation;
    $('detailList').innerHTML=copy.details.map(([c,text])=>`<div class="detail"><i style="--detail:var(${c})"></i><span>${text}</span></div>`).join('');
    $('stageCallout').textContent=copy.callout;

    $('taskSelect').value=state.task;
    $('cameraSelect').value=String(state.cameraCount);
    $('cameraMeaning').textContent=state.cameraCount===1?t.cameraMeaning1:t.cameraMeaning2;
    $('known3DStrip').textContent=t.known3D;
    $('known3DSource').textContent=t.known3DSource;
    $('rigKnownStrip').textContent=state.cameraCount===2?'K₁, K₂ and the fixed camera-to-camera transform are known.':'One calibrated camera matrix K is used.';
    $('unknownStrip').textContent=t.unknown;

    $('pointSourceTitle').textContent=`Where the ${t.short} 3D points come from`;
    $('pointSourceText').textContent=t.pointSource;
    $('eyeLinkTitle').textContent=t.eyeLinkTitle;
    $('eyeLinkText').textContent=t.eyeLinkText;
    $('eyeLinkCallout').textContent=t.eyeLinkCallout;

    $('modeMetric').textContent=state.cameraCount===1?'Classical PnP':'Joint rig PnP';
    $('modeMetricSub').textContent=state.cameraCount===1?'one optical center':'two calibrated optical centers';
    $('pointMetric').textContent=points().length;
    const matches=visibleMatchCount();
    $('matchMetric').textContent=matches;
    $('constraintMetric').textContent=`${matches*2} scalar constraints for 6 unknowns`;

    sliderDefs.forEach(([key,,,,step,unit])=>{
      $(`${key}Slider`).value=state[key];
      $(`${key}Out`).textContent=state[key].toFixed(step<1?2:0)+unit;
    });
    $('baselineRow').style.display=state.cameraCount===2?'grid':'none';
    $('occlusionToggle').checked=state.occlusion;
    $('outlierToggle').checked=state.outlier;
    $('robustToggle').checked=state.robust;
    $('rayToggle').checked=state.rays;
    $('tourBtn').textContent=state.tour?'■ Stop tour':'▶ Guided tour';

    const noteParts=[];
    if(state.outlier) noteParts.push(state.robust?'The large wrong match will be down-weighted during refinement.':'The wrong match has full least-squares influence and may pull the pose away.');
    if(state.occlusion) noteParts.push('Camera 1 has two fewer correspondences.');
    if(state.cameraCount===2) noteParts.push('Camera 2 still constrains the same six rig-pose unknowns.');
    if(!noteParts.length) noteParts.push('Change noise, occlusion, or matching, then compare one camera with two cameras.');
    $('experimentNote').textContent=noteParts.join(' ');
    $('experimentNote').classList.toggle('warning',state.outlier&&!state.robust);

    updatePointTable();
    updatePipeline();
  }
