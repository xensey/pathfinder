  function task7Switch() {
    startTask(7);
    renderShell(7, `
      <div class="stage-kicker">Когнитивная гибкость</div>
      <h2>Классификация с изменением правила</h2>
      <p class="task-instruction">Будет 16 коротких проб. Первые 8 — классификация по форме, затем правило изменится на классификацию по цвету. Текущее правило будет показано над стимулом.</p>
      <div class="note">Старайтесь отвечать быстро, но без угадывания.</div>
      <div class="row" style="margin-top:20px"><button class="btn" id="beginSwitch">Начать</button></div>`);
    document.getElementById('beginSwitch').addEventListener('click', run);

    function run() {
      const trials=[];
      for(let i=0;i<16;i++){
        const phase=i<8?'shape':'color';
        const shape=Math.random()<.5?'circle':'triangle';
        const color=Math.random()<.5?'blue':'orange';
        const correct = phase==='shape' ? (shape==='circle'?'left':'right') : (color==='blue'?'left':'right');
        trials.push({phase,shape,color,correct});
      }
      let i=0; state.switchLog=[];
      const renderTrial=()=>{
        if(i>=trials.length){
          const correctN=state.switchLog.filter(x=>x.correct).length;
          const postSwitchErrors=state.switchLog.slice(8,12).filter(x=>!x.correct).length;
          addResult('switch','Переключение правила',correctN,16,{trials:state.switchLog,postSwitchErrors});
          task8Information(); return;
        }
        const t=trials[i];
        const phaseTitle=t.phase==='shape'?'Правило: КРУГ → левая кнопка, ТРЕУГОЛЬНИК → правая':'Новое правило: СИНИЙ → левая кнопка, ОРАНЖЕВЫЙ → правая';
        renderShell(7, `
          <div class="stage-kicker">Когнитивная гибкость</div>
          <div class="mini-progress"><div style="width:${(i/16)*100}%"></div></div>
          <h2 class="center" style="margin-top:18px">${phaseTitle}</h2>
          ${i===8?'<div class="note warning center"><strong>Правило изменилось</strong></div>':''}
          <div class="stimulus"><div class="shape ${t.shape} ${t.color}"></div></div>
          <div class="response-row"><button class="btn response-btn" id="leftBtn">Левая</button><button class="btn response-btn" id="rightBtn">Правая</button></div>`);
        const shownAt=performance.now();
        const respond=(choice)=>{
          const rt=msToSec(performance.now()-shownAt);
          state.switchLog.push({trial:i+1,phase:t.phase,shape:t.shape,color:t.color,choice,correct:choice===t.correct,rtSec:rt});
          i++; setTimeout(renderTrial,120);
        };
        document.getElementById('leftBtn').addEventListener('click',()=>respond('left'));
        document.getElementById('rightBtn').addEventListener('click',()=>respond('right'));
      };
      renderTrial();
    }
  }
