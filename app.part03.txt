    document.getElementById('showSeq').addEventListener('click', () => {
      renderShell(2, `<div class="stage-kicker">Рабочая память</div><h2 class="center">Смотрите на центр экрана</h2><div class="stimulus"><div id="seqSymbol" class="sequence-symbol"></div></div>`);
      const el = document.getElementById('seqSymbol');
      const showNext = () => {
        if (idx >= state.sequence.length) { setTimeout(renderReverseInput, 550); return; }
        el.textContent = state.sequence[idx];
        idx++;
        setTimeout(()=>{ el.textContent=''; setTimeout(showNext,230); }, 720);
      };
      showNext();
    });

    function renderReverseInput() {
      let answer = [];
      renderShell(2, `
        <div class="stage-kicker">Рабочая память</div>
        <h2>Введите последовательность в обратном порядке</h2>
        <div class="answer-strip" id="answerStrip"><span class="muted">Выбранные символы появятся здесь</span></div>
        <div class="symbol-grid">${state.sequence.map(s=>`<button class="symbol-btn" data-s="${s}">${s}</button>`).join('')}</div>
        <div class="row" style="margin-top:18px"><button class="btn secondary" id="undo">Отменить последний</button><button class="btn" id="submit" disabled>Ответить</button></div>`);
      const strip = document.getElementById('answerStrip');
      function redraw() {
        strip.innerHTML = answer.length ? answer.map(s=>`<span class="answer-chip">${s}</span>`).join('') : '<span class="muted">Выбранные символы появятся здесь</span>';
        document.getElementById('submit').disabled = answer.length !== state.sequence.length;
      }
      document.querySelectorAll('.symbol-btn').forEach(btn => btn.addEventListener('click', ()=>{
        if (answer.length < state.sequence.length) { answer.push(btn.dataset.s); redraw(); }
      }));
      document.getElementById('undo').addEventListener('click', ()=>{ answer.pop(); redraw(); });
      document.getElementById('submit').addEventListener('click', ()=>{
        const correct = [...state.sequence].reverse();
        const correctPositions = answer.filter((x,i)=>x===correct[i]).length;
        addResult('reverse','Обратная последовательность',correctPositions,6,{sequence:state.sequence,answer,correct});
        task3Relations();
      });
    }
  }
