  function shuffle(arr) {
    const a = [...arr];
    for (let i=a.length-1;i>0;i--) {
      const j = Math.floor(Math.random()*(i+1));
      [a[i],a[j]]=[a[j],a[i]];
    }

    return h > 0 ? `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}` : `${m}:${String(s).padStart(2,'0')}`;
  }

  function cameraStatusCopy() {
    const status = state.camera.status;
    if (status === 'requesting') return 'Ожидается разрешение браузера…';
    if (status === 'ready') return 'Камера подключена. Запись начнётся вместе с тестом.';
    if (status === 'recording') return 'Идёт видеозапись.';
    if (status === 'denied') return 'Доступ к камере отклонён.';
    if (status === 'unsupported') return 'Этот браузер не поддерживает запись камеры.';
    if (status === 'error') return 'Камеру не удалось подключить.';
    if (status === 'skipped') return 'Тест запущен без видеозаписи.';
    return 'Камера ещё не подключена.';
  }
